program main{
    var i,j,count : int;
    begin;
        i := 0;
        count := 0;
        while (i < 2) do
        {
            j := 0;
            while (j < 2) do
            {
                count := count + 1;
                j++;
            }
            i++;
        }
        write(count);
    end;
}
