program main {
    var check_a, check_b : bool;
    var counter : int;
    begin;
        write("Testing dangling else structural integrity");
        check_a := true;
        check_b := false;
        counter := 0;

        if (check_a) then {
            if (check_b) then {
                counter := 1;
            }
        } else {
            counter := 2; 
        }
        
        write(counter);
    end;
}
